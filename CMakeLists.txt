cmake_minimum_required(VERSION 3.16)
project(mini_encrypted_vpn VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_SANITIZERS "Compile with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(BUILD_FUZZING "Build the libFuzzer packet decoder harness (Clang only)" OFF)
if(ENABLE_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizer build requires GCC or Clang")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()
find_package(OpenSSL REQUIRED)
include_directories(include)

add_executable(vpn_core src/main.cpp)
add_library(tunnel_crypto STATIC src/crypto.cpp src/handshake.cpp)
target_link_libraries(tunnel_crypto PUBLIC OpenSSL::Crypto)

add_executable(vpn_server src/server.cpp src/tun.cpp)
target_link_libraries(vpn_server PRIVATE tunnel_crypto)
add_executable(vpn_client src/client.cpp src/tun.cpp)
target_link_libraries(vpn_client PRIVATE tunnel_crypto)
add_executable(tun_test src/tun_test.cpp)

include(CTest)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(session_test tests/session_test.cpp)
    target_link_libraries(session_test PRIVATE tunnel_crypto)
    add_test(NAME session_packets COMMAND session_test)
    add_executable(handshake_test tests/handshake_test.cpp)
    target_link_libraries(handshake_test PRIVATE tunnel_crypto Threads::Threads)
    add_test(NAME handshake_auth COMMAND handshake_test)
    set_tests_properties(handshake_auth PROPERTIES TIMEOUT 25)
    add_executable(replay_study evaluation/replay_study.cpp)
    add_test(NAME replay_window_model COMMAND replay_study)
    add_executable(replay_robustness evaluation/replay_robustness.cpp)
    add_test(NAME replay_robustness_campaign COMMAND replay_robustness)
    set_tests_properties(replay_robustness_campaign PROPERTIES TIMEOUT 45)

    add_executable(session_campaign_test tests/session_campaign_test.cpp)
    target_link_libraries(session_campaign_test PRIVATE tunnel_crypto)
    add_test(NAME packet_attack_campaign COMMAND session_campaign_test)
    add_executable(tun_ready_test tests/tun_ready_test.cpp src/tun.cpp)
    target_link_libraries(tun_ready_test PRIVATE Threads::Threads)
    add_test(NAME tun_startup_readiness COMMAND tun_ready_test)
endif()

if(BUILD_FUZZING)
    if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        message(FATAL_ERROR "libFuzzer target requires clang++")
    endif()
    add_executable(packet_fuzzer fuzz/packet_fuzzer.cpp)
    target_compile_options(packet_fuzzer PRIVATE -fsanitize=fuzzer)
    target_link_options(packet_fuzzer PRIVATE -fsanitize=fuzzer)
    target_link_libraries(packet_fuzzer PRIVATE tunnel_crypto)
endif()
